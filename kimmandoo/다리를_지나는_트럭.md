# 접근 방식
queue를 좀 더 유연하게 쓰자
처음에는 지나고있는 다리를 그냥 cnt로 값 더해서 처리하려고했는데, 이거 너무 어렵다 -> 큐 2개

# 학습 내용

지나는 다리도 q로 해주면, 계산하기 훨씬 쉬워진다

# 풀이 코드 
```java
import java.util.*;

class Solution {
    public int solution(int len, int w, int[] trucks) {
        // 다리를 나타내는 큐
        Queue<Integer> bridge = new LinkedList<>();
        // 대기 트럭 큐
        Queue<Integer> q = new LinkedList<>();
        for (int truck : trucks) {
            q.offer(truck);
        }

        // 다리 길이만큼 0으로 채워 초기화
        for (int i = 0; i < len; i++) {
            bridge.offer(0);
        }

        int time = 0;
        int curW = 0; // 현재 다리 위 무게

        // 대기 트럭이 없을 때까지 반복
        while (!q.isEmpty()) {
            time++;

            // 1. 다리에서 한 칸 이동 (맨 앞 요소 제거)
            curW -= bridge.poll();

            // 2. 새 트럭이 진입 가능한지 확인
            if (curW + q.peek() <= w) {
                int truck = q.poll();
                bridge.offer(truck);
                curW += truck;
            } else {
                // 진입 불가 시, 0을 넣어 빈 공간으로 채움
                bridge.offer(0);
            }
        }

        return time + len; // len만큼 더해주는 이유는 다리만큼 지나는 시간이 무조건 포함되기 때문
    }
}
```
