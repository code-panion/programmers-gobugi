# 접근 방식
큐 문제인데, 따로 class로 객체를 관리해야된다고 판단했다.

idx랑 우선순위를 같이 들고 다녀서 q를 계속 돌면서 판단하는 문제

# 학습 내용

오랜만에 설계한대로 쭉 풀려서 기분 좋다

# 풀이 코드 
```java
import java.util.*;

class Solution {
    static class Node {
        int idx, pr;
        Node(int i, int p) {
            idx = i;
            pr = p;
        }
    }

    public int solution(int[] pr, int l) {
        int answer = 0;
        int[] freq = new int[10];
        for (int i : pr) {
            freq[i]++;
        }

        Queue<Node> q = new LinkedList<>();
        for (int i = 0; i < pr.length; i++) {
            q.offer(new Node(i, pr[i]));
        }

        int cnt = 0;
        while (!q.isEmpty()) {
            Node cur = q.poll();
            boolean high = false;

            for (int i = cur.pr + 1; i < 10; i++) {
                if (freq[i] > 0) {
                    high = true;
                    break;
                }
            }

            if (high) {
                q.offer(cur);
            } else {
                freq[cur.pr]--;
                cnt++;
                if (cur.idx == l) {
                    return cnt;
                }
            }
        }

        return -1; // 여기에 도달하면 안 됨
    }
}
```
