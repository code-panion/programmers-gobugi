# 접근 방식

각 사람마다 패턴이 있어서, 그 패턴을 미리 저장해서 사용했다.
패턴과 계속 일치하면 점수를 늘려주고, 중복으로 공동 1등할 수 있어서 차례로 검사 후 추가해줬다.

# 학습 내용


# 풀이 코드 
```java
import java.util.*;

class Solution {
    public int[] solution(int[] answers) {
        int[] a = new int[]{1,2,3,4,5};
        int[] b = new int[]{2,1,2,3,2,4,2,5};
        int[] c = new int[]{3,3,1,1,2,2,4,4,5,5};
        ArrayList<Integer> arr = new ArrayList();
        int max, aa, bb, cc;
        max = aa = bb = cc = 0;
        for(int i=0; i<answers.length; i++){
            int tmp = answers[i];
            if(tmp == a[i%5]) aa++;
            if(tmp == b[i%8]) bb++;
            if(tmp == c[i%10]) cc++;
            if(max < aa) max = aa;
            if(max < bb) max = bb;
            if(max < cc) max = cc;
        }

        if(aa == max) arr.add(1);
        if(bb == max) arr.add(2);
        if(cc == max) arr.add(3);
        
        int[] answer = new int[arr.size()];
        int idx = 0;
        
        for(int i: arr){
            answer[idx++] = i;
        }
        
        return answer;
    }
}
```
