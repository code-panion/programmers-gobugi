# 접근 방식
일단 노란 타일의 갯수를 보고 가로가 세로보다 길거나 같은 노란 타일의 가로, 세로 조합을 완전 탐색했다. 그런 다음 노란 타일 기준으로 카펫의 전체 사이즈를 구할 수 있게 que 를 돌려서 구했다.

# 학습내용
갈색 타일은 노란 타일을 가운데에 두고 두르기 때문에 케이스 1을 참고하면 노란 타일이 w, h 가 2,1 일때 
갈색 타일의 가로 세로, 즉 bw 와 bh 는 노란타일을 감싸야 하기 떄문에 4, 3 => 한번 더 두르면 6, 5 => 한번 더 감싸면 8, 7 의 길이를 가지게 된다. 
이를 통해 갈색타일의 가로, 세로 길이는 중심으로 부터 멀어질 때마다 2*n 만큼 늘어나는 걸 알 수있다. 
그래서 갈색타일의 갯수는 그런 가로, 세로를 기준으로 구하되, 중복되는 네 모서리를 제외하도록 4를 빼서 처리했다.

# 풀이 코드

```
y_check = [0] * 2000001

def solve(lst, cnt):
    while lst:
        w, h = lst.pop(0)
        n = 1
        is_correct = 0
        #print(w,h)
        while is_correct < cnt:
            is_correct += (2*n+w) * 2 + (2*n+h) * 2 - 4
            #print(is_correct)
            n += 1
        if is_correct == cnt:
            bw = (2*(n-1)+w)
            bh = (2*(n-1)+h)
            return [bw, bh]

# 노란 타일 가로 세로 조합 찾기 가로 >= 세로 
def y_comb(num):
    que = []
    for i in range(1, num + 1):
        if not y_check[i] and not num % i:
            y_check[i] = 1
            y_check[num//i] = 1
            w = max(i, num //i)
            h = min(i, num //i)
            que.append((w, h))
    return que

def solution(brown, yellow):
    answer = []
    lst = y_comb(yellow)
    #print(lst)
    answer = solve(lst, brown)
    return answer
```
