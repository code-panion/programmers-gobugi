# 접근 방식
가장 많이 재생된 장르를 우선 순위로, 
해당 장르별 베스트 곡 즉, 많이 재생된 곡 두 곡의 i 값을 return 되도록 구현하였다. 
이때 같은 횟수로 재생됐다면 i 값이 작은 수가 먼저 출력되도록 구현한다. 


</br>
</br>

# 학습내용
우선순위 큐를 쓸까 하다가 딕셔너리를 sort 하는 방법이 있길래 참고하여 작성하였다.

</br>

# 풀이 코드

```
from collections import defaultdict
import queue

def solution(genres, plays):
    answer = []
    repeat_lst = defaultdict(list)
    total_cnt = defaultdict(int)
    genre_lst = defaultdict(list)
    for i in range(len(genres)):
        total_cnt[genres[i]] += plays[i]
        repeat_lst[plays[i]].append(i)
        genre_lst[genres[i]].append(plays[i])
    prior_gen = sorted(total_cnt.items(), key = lambda x: x[1], reverse = True)
    for key in genre_lst.keys():
        genre_lst[key].sort()
    #print(genre_lst)
    for i in range(len(prior_gen)):
        key, value = prior_gen[i]
        cnt = 0
        flg = True
        if genre_lst[key]:
            while cnt < 2 and flg:
                for i in range(len(genre_lst[key])-1, -1, -1):
                    num = genre_lst[key][i]
                    s = 0
                    while s < len(repeat_lst[num]) and flg:
                        answer.append(repeat_lst[num][s])
                        cnt += 1
                        if cnt == 2:
                            flg = False
                            break
                        elif cnt == len(genre_lst[key]):
                            flg = False
                            break
                        s += 1
                    
    return answer
```
