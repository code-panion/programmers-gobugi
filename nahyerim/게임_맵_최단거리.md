# 접근 방식
bfs 방식으로 푸는데, 각 타일이 벽이 없고 지나갈 수 없다면 해당 타일을 최단으로 갈 수 있는 이동 걸음을 체크하도록 했다.

# 풀이 코드

```
def solution(maps):
    answer = 0
    # 상, 하, 좌, 우 
    dr = [-1, 1, 0, 0]
    dc = [0, 0, -1, 1]
    tr, tc = len(maps) - 1, len(maps[0]) - 1
    que = [(0, 0, 1)]
    visited = [[0] * (tc+1) for row in range(tr+1)]
    visited[0][0] = 1
    while que:
        cr, cc, m = que.pop(0)
        if cr == tr and cc == tc:
            answer = m 
        for i in range(4):
            nr = cr + dr[i]
            nc = cc + dc[i]
            if 0 <= nr <= tr and 0 <= nc <= tc and maps[nr][nc]:
                if not visited[nr][nc]:
                    visited[nr][nc] = m + 1
                    que.append((nr, nc, m + 1))
                elif visited[nr][nc] and m + 1 < visited[nr][nc]:
                    visited[nr][nc] = m + 1
                    que.append((nr,nc, m + 1))
    if not answer:
        answer = -1
                
    return answer
```
