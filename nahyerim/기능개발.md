# 풀이 코드

```
def how_days(i, speed):
    cnt = 1
    task = i
    while task  < 100:
        task = i + speed * cnt
        cnt += 1
    return cnt

def solution(progresses, speeds):
    ing = 0 
    day = 0
    answer = []
    k = 1
    for i in range(len(progresses)):
        if not ing:
            ing = progresses[i]
            day = how_days(progresses[i], speeds[i])
        else:
            after = how_days(progresses[i], speeds[i])
            if day >= after:
                k += 1
            else:
                answer.append(k)
                ing = progresses[i] 
                day = after
                k = 1
    answer.append(k)            
    #print(answer)
    return answer
```
