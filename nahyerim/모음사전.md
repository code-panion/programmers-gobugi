# 접근 방식
완전 탐색이라서 모음을 우선순위대로 삽인한 list vowel 을 완전 탐색 돌리면서 해당 word가 되는 순간 값을 return 하도록 처리했더니 해결됨.
x 는 해당 배열의 요소 번호, comb는 만들어지는 단어, v는 방문체크, i는 단어길이로 생각하여 코드를 구성함.

# 풀이 코드

```
answer = 0
val = 0

def find_word(x, comb, v, i, word):
    global answer
    global val 
    vowel = ["A", "E", "I", "O", "U"]
    if not answer:
        comb += vowel[x]
        val += 1
        if word == comb:
            answer = val
            return 
        v[i][x] = 1
        if i == 4:
            return 
        for k in range(5):
            if not v[i+1][k]:
                v[i+1][k] = 1
                find_word(k, comb, v, i + 1, word)
                v[i+1][k] = 0
    else:
        return 
        
                
    


def solution(word):
    global answer 
    vowel = ["A", "E", "I", "O", "U"]
    check = [[0] * 5 for row in range(5)]
    for i in range(5):
        find_word(i, "", check, 0, word)
    
    return answer
```
