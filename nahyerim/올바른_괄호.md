# 접근 방식
딱 스택을 응용한 문제라고 생각해서 꺼낸 문자열과 이전에 스택에 집어넣은 문자열들을 비교하도록 코드를 구현하면 되겠다고 생각함. 


</br>
</br>

# 학습내용
파이썬 내 덱으로 구현하면 시간이 더 짧아질까 생각했지만... 그냥 리스트.pop()으로 해도 통과되길래 일단 그대로 구현했음. 오랜만에 반복문을 돌리려니 
조건문이 어색해서 많이 풀어봐야겠다고 느낌. 

</br>
</br>

# 풀이 코드

```
def solution(s):
    answer = True
    
    # [실행] 버튼을 누르면 출력 값을 볼 수 있습니다.
    check = []
    for i in s:
        # 만약 체크가 비였는데 여는 괄호로 시작되면
        if not check and i == "(":
            check.append(i)
        # 만약 체크가 비였는데 여는 괄호로 시작되지 않으면    
        elif not check and i == ")":
            answer = False
            break
        # 만약 바로 직전에 들어간 괄호가 여는 괄호고 현재 괄호가 닫는 괄호면 체크 안 여는 괄호 제거 
        elif check[-1] != i and check[-1] == "(":
            check.pop()
        # 만약 바로 직전에 들어간 괄호가 닫는 괄호고 현재 괄호가 여는 괄호면  
        elif check[-1] != i and check[-1] == ")":
            cnt = 0
            start = -2
            # 여는 괄호와 닫는 괄호가 짝맞춰 존재하는지 체크 
            while -len(check) <= start < -1:
                if check[start] == check[-1]:
                    start -= 1
                cnt += 1
                start += 1
            # 그리고 짝맞춰 존재하는 만큼 체크에서 없애기 
            while cnt:
                check.pop()
                cnt -= 1
        # 저런 경우가 아니라면 다 체크에 집어넣기 
        else:
            check.append(i)
    # 만약 체크가 남았다면 짝이 안 남는 거기 때문에 실패 
    if check:
        answer = False
    return answer
```
