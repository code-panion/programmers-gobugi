# 접근 방식
완전 탐색이라 어떤 식으로 순회할 지와 가지치기를 어떻게 할지 고민을 많이 했다. 지갑의 가로, 세로는 바꿔서 생각할 수 있되, 다른 명함들의 가로, 세로 중 한 값보단 크거나 같은 값들로 구현하여야 명함들이 다 들어가는 지갑을 만들 수 있다는 것에 주목하여 코드를 짰다.

# 학습내용
3문제 정도 시간 초과 나길래 어디서 더 줄일 수 있을까 봤더니, 명함의 가로 세로 사이즈는 중복될 수 있으니 중복된 길이는 체크하지 않도록 넘기니 통과할 수 있었다. solve_w 와 solve_h 는 예외처리 하는데서 코드가 길어지길래 나눴는데 가로인지 세로인지 구분하는 변수를 하나 더 추가하면 함수를 줄일 수 있을 것 같다.

# 풀이 코드
개선 후 코드 
: solve_w 와 solve_h를 가로와 세로 중 어느걸 구하는 지 확인하기 위한 배열 dr을 매개변수로 바꿔서 코드를 더 깔끔하게 만들었다. 
dr[0] 일땐 가로길이를 구하고 dr[1]일땐 세로길이를 구하도록 했다. 그리고 for 문 내에서 해당 길이가 유효한가를 확인하는 if문을 
min 과 max 함수를 이용해 더 깔끔하게 만들었다.  

```
min_w = 1001
min_h = 1001
min_v = 999999999999999999999999999999999999999

def solve(idx, val, arr, dr):
    global min_w 
    global min_h 
    global min_v
    if dr[0] and min_w < val:
        return 
    if dr[1] and min_h < val:
        return 
    cnt = 0
    if dr[0]:
        other_dr = min_h
    elif dr[1]:
        other_dr = min_w 
    for i in range(len(arr)):
        if i != idx:
            if min(arr[i]) > val:
                return 
            elif val >= min(arr[i]) and other_dr >= max(arr[i]) or val >= max(arr[i]):
                cnt += 1
                
    if dr[0] and cnt >= len(arr) - 1 and val < min_w:
        min_w = val 
        return 
    elif dr[1] and cnt >= len(arr) - 1 and val < min_h:
        min_h = val 
        return 
    
def solution(sizes):
    answer = 0
    global min_v
    global min_w
    global min_h
    used = [0] * 1001
    for i in range(len(sizes)):
        if not used[sizes[i][0]]:
            used[sizes[i][0]] = 1
            solve(i, sizes[i][0], sizes, [1,0])
            solve(i, sizes[i][0], sizes, [0,1])
        if not used[sizes[i][1]]:
            used[sizes[i][1]] = 1
            solve(i, sizes[i][1], sizes, [1,0])
            solve(i, sizes[i][1], sizes, [0,1])
    #print(min_w, min_h)
    answer = min_w * min_h
    return answer
```

#최소 가로길이
min_w = 1001

min_h = 1001

# 지갑의 최소 세로 길이를 구하는 함수 
def solve_h(idx, val, arr):
    global min_w 
    global min_h 
    global min_v 
    if min_h < val:
        return 
    cnt_h = 0
    for i in range(len(arr)):
        if i != idx:
        # 현재 길이가 다른 명함들의 가로, 세로 중 한 가지 이상은 크되, 현재 최소 가로길이가 val 보다 큰 값을 커버쳐야 올바른 지갑 사이즈를 만들 수 있다.
        # 그래서 그런 조건을 충족하는지 확인하는 코드를 짬. 
            a, b = arr[i][0], arr[i][1]
            if a > val and b > val:
                return 
            elif a <= val and b <= val:
                cnt_h += 1
            elif a < val and b <= val and a <= min_w:
                cnt_h += 1
            elif b < val and a <= val and b <= min_w:
                cnt_h += 1
                
    if cnt_h >= len(arr) - 1 and val < min_h:
        min_h = val
        return 


# 지갑의 최소 가로 길이를 구하는 함수 
def solve_w(idx, val, arr):
    global min_w 
    global min_h 
    global min_v 
    if min_w < val:
        return 
    cnt_w = 0
    for i in range(len(arr)):
        # 현재 길이가 다른 명함들의 가로, 세로 중 한 가지 이상은 크되, 현재 최대 세로길이가 val 보다 큰 값을 커버쳐야 올바른 지갑 사이즈를 만들 수 있다.
        # 그래서 그런 조건을 충족하는지 확인하는 코드를 짬. 
        if i != idx:
            a, b = arr[i][0], arr[i][1]
            if a > val and b > val:
                return 
            elif a <= val and b <= val:
                cnt_w += 1
            elif a > val and b <= val and a <= min_h:
                cnt_w += 1
            elif b > val and a <= val and b <= min_h:
                cnt_w += 1
                
    if cnt_w >= len(arr) - 1 and val < min_w:
        min_w = val 
        return 
    
def solution(sizes):
    answer = 0
    global min_v
    global min_w
    global min_h
    # 명함 길이 방문체크용 
    used = [0] * 1001
    for i in range(len(sizes)):
        if not used[sizes[i][0]]:
            used[sizes[i][0]] = 1
            solve_w(i, sizes[i][0], sizes)
            solve_h(i, sizes[i][0], sizes)
        if not used[sizes[i][1]]:
            used[sizes[i][1]] = 1
            solve_w(i, sizes[i][1], sizes)
            solve_h(i, sizes[i][1], sizes)
    #print(min_w, min_h)
