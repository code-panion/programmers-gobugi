# 접근 방식
sort() 함수를 사용해서 큰 값을 먼저 붙이려고 했지만 그러면 30,3,34 같은 값들이 처리되지 않아 고민을 엄청했다. 그래서 각 자릿수를 비교하여
작은 자릿수의 숫자가 클 수록 우선으로 붙도록 처리하려고 생각했는데 그러면 어쩔 수 없이 앞자리 수가 같은 list 를 그 리스트의 길이 만큼 순회해야 해서 
시간 초과로 터짐... 그래서 풀이법을 참고하여 해결하였다.

# 학습내용
3, 30, 34 같은 값들을 어떻게 우선순위를 처리할 지 엄청 고민했는데 그 값들을 3배로 늘려주면, 즉 333, 303030, 343434 이렇게 만들어준뒤 sort 시키면 원하는 대로 만들어진다는 풀이를 보고 감탄했음. 마지막 int 처리는 00 같은 값들을 없애기 위한 거라고 한다.

# 풀이 코드

### 참고한 코드
```
def solution(numbers):
    answer = ''
    
    numbers = list(map(str, numbers))               # string으로 바꾸어서 비교
    numbers.sort(key = lambda x : x*3,reverse=True)  # 3번씩 반복하면 붙였을때 큰수 찾기 가능
    
    for i in numbers:             # 정렬된 리스트를 answer에 순서대로 더해줌
        answer += i
    
    return str(int(answer))
```

def solve(num, arr):
    sort_arr = ''
    cnt = 0
    used = [0] * len(arr)
    #print(arr)
    while cnt < len(arr):
        max_v = -1
        max_i = -1
        for i in range(len(arr)):
            if not used[i]:
                if max_v == -1 and max_i == -1:
                    max_v = int(arr[i])
                    max_i = i
                for j in range(len(str(arr[i]))-1):
                    for k in range(j+1, len(str(arr[i]))):
                        if int(arr[i][k]) >= num and int(arr[i][k]) > int(arr[i][j]):
                            max_v = int(arr[i])
                            max_i = i
        #print(max_i, max_v)
        sort_arr += arr[max_i]
        #print(sort_arr)
        used[max_i] = 1
        cnt += 1
    return sort_arr
