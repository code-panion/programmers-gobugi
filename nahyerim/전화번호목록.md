# 학습내용
챗지랑 이야기하다가 sort 하면 번호순으로 정렬되고, startwith로 인접한 인덱스의 값을 비교하면 더 빠르다 길래 그렇게 하니 해결함.

### 📌 startswith함수란? 
프로그래밍에서 특정 문자열이 다른 문자열로 시작하는지 확인하는 데 사용되는 메서드 </br>
```
str = 'final exam'
# startswith(특정 문자)
result = str.startswith('final')
print(result) //True
```

# 풀이 코드

개선 코드 

```
def solution(phone_book):
    answer = True
    phone_book.sort() 
    for i in range(len(phone_book) - 1):
        if phone_book[i+1].startswith(phone_book[i]):
            return False
    return True    
```

딕셔너리 이용 버전
```
def solution(phone_book):
    phone_dict = {}
    for number in phone_book:
        phone_dict[number] = 1

    for number in phone_book:
        prefix = ""
        for char in number[:-1]:  # 마지막 문자 제외
            prefix += char
            if prefix in phone_dict:
                return False
    return True
```

```
def solve(i, j, arr):
    flg = True
    end = min(len(arr[i]), len(arr[j]))
    s = 0
    cnt = 0
    while s < end:
        if arr[i][s] == arr[j][s]:
            cnt += 1
        else:
            break
        s+=1
    if cnt == end:
        flg = False
        #print("실패")
        return flg
    return flg

def solution(phone_book):
    answer = True
    phone_book.sort() 
    for i in range(len(phone_book)):
        if answer:
            for j in range(i+1, len(phone_book)):
                result = solve(i, j, phone_book)
                if not result:
                    answer = False
                    break
    return answer

```
