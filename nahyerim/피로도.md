# 접근 방식
완전 탐색이라서 현재 k 값 또한 매개변수로 넣고 최소 소모 피로도랑 소모 피로도보다 현재 k 값이 크거나 같을 때 던전을 돌도록 설정했다.

# 풀이 코드

```
answer = - 1 
def brute_force(idx, arr, v, k):
    global answer 
    if idx > answer:
        answer = idx
    if idx == len(arr):
        return 
    for i in range(len(arr)):
        if not v[i] and arr[i][0] <= k and arr[i][1] <= k:
            v[i] = 1
            brute_force(idx+1, arr, v, k - arr[i][1])
            v[i] = 0

def solution(k, dungeons):
    global answer 
    v = [0] * len(dungeons)
    for i in range(len(dungeons)):
        brute_force(0, dungeons, v, k)
    return answer
```
