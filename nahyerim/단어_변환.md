# 접근 방식
 begin 을 현재 단어로 시작해서 변환하는 과정을 거칠 때 words 리스트 안 단어들 중 철자가 딱 한 개가 다른경우만 변환 시킬 수 있기 때문에 그러한 조건의 
 단어를 방문체크하면서 que에 넣되, 만약 방문했어도 현재 방문체크됐던 변환 횟수보다 작은 횟수로 변경이 가능하다면 다시 집어넣도록 했다.

# 풀이 코드

```
def solution(begin, target, words):
    answer = 0
    que = [(begin, 0)]
    v = [0] * len(words)
    while que:
        word, step = que.pop(0)
        if word == target:
            if not answer:
                answer = step
            elif answer and answer > step:
                answer = step 
        for i in range(len(words)):
            if words[i] != word:
                s = 0
                cnt = 0
                flg = True
                while s < len(begin):
                    if word[s] != words[i][s]:
                        cnt += 1
                    if cnt > 1:
                        flg = False
                        break
                    s += 1
                if flg and not v[i]:
                    v[i] = step + 1
                    que.append((words[i], step + 1))
                elif flg and v[i] and v[i] > step + 1:
                    v[i] = step + 1
                    que.append((words[i], step + 1))    
        
    return answer
```
