# 접근 방식
이어진 전선 중 하나만 끊어서 전력망을 나누는 것이기 때문에 그룹 2개를 나눠서 각 그룹마다 해당 숫자의 송전탑이 있는지 확인하는 방식으로 구현했다. 


</br>
</br>

# 학습내용
트리 같은 구조이므로 1부터 n까지의 숫자는 모두 어떤 그룹 안에 반드시 속해있어야 한다. 그래서 net1과 net2 그룹을 설정하고 끊을 송전탑 번호로 net1과 net2에 각각 해당 송전탑이 있다고 표시했다. 그리고 다른 전선정보는 que에 넣은 후 net1 과 net2에 있는 송전탑 넘버와 이어진 전선은 해당 그룹임으로 해당 전산망에 이어진 송전탑 번호를 넣고, 아직 그룹망에 포함되지 않는 (선조와 연결되지 않는) 전선탑의 경우 다시 que에 넣도록 하여 구현했다. 
</br>
</br>

# 풀이 코드

```
answer = 101

def network(idx, arr):
    global answer
    if answer == 0:
        return 
    net1 = [0] * (len(arr)+2)
    net1[arr[idx][0]] = 1
    net2 = [0] * (len(arr)+2)
    net2[arr[idx][1]] = 1
    que = []
    for i in range(len(arr)):
        if i != idx:
            que.append((arr[i][0], arr[i][1]))
    while que:
        v1, v2 = que.pop(0)
        if net1[v1]:
            net1[v2] = 1
        elif net2[v1]:
            net2[v2] = 1
        if net1[v2]:
            net1[v1] = 1
        elif net2[v2]:
            net2[v1] = 1
        if not net1[v1] and not net1[v2] and not net2[v1] and not net2[v2]:
            que.append((v1, v2))
    cnt1 = 0
    cnt2 = 0
    s = 1
    while s < len(arr)+2:
        if net1[s]:
            cnt1 += 1
        elif net2[s]:
            cnt2 += 1
        s+= 1
    if answer > max(cnt1, cnt2) - min(cnt1, cnt2):
        answer = max(cnt1, cnt2) - min(cnt1, cnt2)
    return 
    
def solution(n, wires):
    global answer 
    for i in range(len(wires)):
        network(i, wires)
    return answer
```
