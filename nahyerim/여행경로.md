# 접근 방식

# 학습내용
테스트 케이스 중 첫번째 문제만 틀리길래 왜 그런 걸까, 하고 봤더니 공항의 첫글자만 알파벳을 비교하고 있었다는 사실을 깨닫고 모든 글자를 비교하도록
코드를 변환하니 해결됐다.

# 풀이 코드

```
answer = []
def solve(used, tickets, que, idx):
    #print(que, idx)
    global answer 
    if idx == len(tickets):
        if not answer:
            answer = que
        elif answer:
            s = 0
            #print(answer, que)
            flg = True
            while s < len(answer) and flg:
                for i in range(3):
                    if ord(answer[s][i]) > ord(que[s][i]):
                        answer = que
                        flg = False
                        break
                    elif ord(answer[s][i]) < ord(que[s][i]):
                        flg = False
                        break
                s += 1
        return 
    for i in range(len(tickets)):
        if not used[i]:
            a, b = tickets[i]
            if que[-1] == a:
                used[i] = 1
                solve(used, tickets, que + [b], idx + 1)
                used[i] = 0
                
def solution(tickets):
    global answer 
    used = [0] * len(tickets)
    for i in range(len(tickets)):
        a, b = tickets[i]
        if a == "ICN":
            used[i] = 1
            solve(used, tickets, [a, b], 1)
            used[i] = 0
    return answer
```
