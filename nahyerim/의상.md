# 풀이 코드

재귀 + 메모이제이션 사용 
```
from collections import defaultdict
from functools import lru_cache

def solution(clothes):
    groups = defaultdict(list)
    for name, cat in clothes:
        groups[cat].append(name)

    cats = sorted(groups.keys())
    n = len(cats)

    @lru_cache(None)
    def dfs(i, picked_any):
        if i == n:
            return 1 if picked_any else 0
        # 0개 선택
        total = dfs(i + 1, picked_any)
        # 1개 선택(이 카테고리 아이템 수만큼)
        total += len(groups[cats[i]]) * dfs(i + 1, True)
        return total

    return dfs(0, False)
```

수학 공식 사용
(옷을 입는 경우의 수는 n개의 옷가지 중 1개를 선택하는 n 개 + 안 입는다 선택지라)
만약 옵션이 2개라고 한다면
(o1 + 1) * (o2 +1) 
(o1의 갯수만큼 + o1 안입는거) *(o2의 갯수만큼 + o2 안입는거)
이렇게 곱으로 이뤄짐으로 곱셈으로 구하되 마지막에 아무것도 안입는 한 가지 경우를 빼주기 
```
from collections import Counter

def solution(clothes):
    counts = Counter(cat for _, cat in clothes)
    ans = 1
    for c in counts.values():
        ans *= (c + 1)
    return ans - 1
```
